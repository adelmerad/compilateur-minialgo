PROGRAM TestComp

DECL
    // ── Variables simples ──────────────────────────────────────
    INTEGER : A, B, C, D, X, Y, Z ;
    INTEGER : I, J, K, N, M ;
    INTEGER : Max, Min, Res ;
    FLOAT   : F1, F2, F3 ;

    // ── Tableaux ───────────────────────────────────────────────
    INTEGER : T[10] ;
    FLOAT   : TF[5] ;
    INTEGER : MAT[8] ;

    // ── Constantes ─────────────────────────────────────────────
    CONST TAILLE = 10 ;
    CONST PI = 3.14 ;
    CONST ZERO = 0 ;
    CONST SEUIL = (-50) ;

ENDDECL

BEGIN

    /*
    ================================================================
     1. AFFECTATIONS SIMPLES
    ================================================================
    */

    A = 5 ;
    B = 3 ;
    C = 0 ;
    N = 10 ;
    M = 1 ;
    F1 = 3.14 ;
    F2 = 2.71 ;

    // Affectations avec valeurs signees
    D = (-7) ;
    F3 = (-1.5) ;

    /*
    ================================================================
     2. EXPRESSIONS ARITHMETIQUES
    ================================================================
    */

    // Addition
    Res = A + B ;

    // Soustraction
    Res = A - B ;

    // Multiplication
    Res = A * B ;

    // Division
    Res = A / B ;

    // Expression composee
    Res = A + B * C - D ;

    // Avec parentheses
    Res = (A + B) * (C - D) ;

    // Expression avec float
    F3 = F1 + F2 ;
    F3 = F1 * F2 - F2 ;

    // Expression longue
    Res = A + B + C + D + M ;
    Res = (A + B) * (C + D) / M ;

    /*
    ================================================================
     3. TABLEAUX
    ================================================================
    */

    // Affectation dans un tableau
    T[0] = 10 ;
    T[1] = 20 ;
    T[2] = 30 ;
    T[9] = 99 ;

    TF[0] = 1.1 ;
    TF[4] = 9.9 ;

    // Lecture depuis un tableau
    A = T[0] ;
    B = T[1] + T[2] ;
    F1 = TF[0] + TF[4] ;

    // Tableau avec indice variable
    T[I] = A + B ;
    A = T[J] ;

    /*
    ================================================================
     4. IF SIMPLE (sans ELSE)
    ================================================================
    */

    IF (A > 0) {
        Res = A ;
    }

    IF (B != 0) {
        Res = B * 2 ;
    }

    /*
    ================================================================
     5. IF / ELSE SIMPLE
    ================================================================
    */

    IF (A > B) {
        Max = A ;
    }
    ELSE {
        Max = B ;
    }

    IF (A < B) {
        Min = A ;
    }
    ELSE {
        Min = B ;
    }

    // Toutes les conditions de comparaison
    IF (A == B) { C = 1 ; }
    ELSE        { C = 0 ; }

    IF (A != B) { C = 1 ; }
    ELSE        { C = 0 ; }

    IF (A <= B) { C = 1 ; }
    ELSE        { C = 0 ; }

    IF (A >= B) { C = 1 ; }
    ELSE        { C = 0 ; }

    /*
    ================================================================
     6. CONDITIONS COMPOSEES (&&, ||, !)
    ================================================================
    */

    IF (A > 0 && B > 0) {
        Res = A + B ;
    }
    ELSE {
        Res = 0 ;
    }

    IF (A > 0 || B > 0) {
        Res = A + B ;
    }

    IF (!A == 0) {
        Res = A ;
    }

    // Condition complexe avec parentheses
    IF ((A > 0 && B > 0) || C > 0) {
        Res = A + B + C ;
    }
    ELSE {
        Res = (-1) ;
    }

    /*
    ================================================================
     7. IF IMBRIQUE (nested IF)
    ================================================================
    */

    IF (A > 0) {
        IF (B > 0) {
            Res = A + B ;
        }
        ELSE {
            Res = A ;
        }
    }
    ELSE {
        IF (B > 0) {
            Res = B ;
        }
        ELSE {
            Res = 0 ;
        }
    }

    // Triple imbrication
    IF (A > 0) {
        IF (B > 0) {
            IF (C > 0) {
                Res = A + B + C ;
            }
            ELSE {
                Res = A + B ;
            }
        }
    }

    /*
    ================================================================
     8. WHILE SIMPLE
    ================================================================
    */

    I = 0 ;
    WHILE (I < N) {
        Res = Res + I ;
        I = I + 1 ;
    }

    // WHILE avec condition complexe
    A = 0 ;
    B = 10 ;
    WHILE (A < B && A >= 0) {
        A = A + 1 ;
    }

    /*
    ================================================================
     9. FOR SIMPLE
    ================================================================
    */

    // FOR : variable : debut : pas : fin
    FOR (I : 0 : 1 : 9) {
        T[I] = I * 2 ;
    }

    FOR (J : 0 : 1 : 4) {
        TF[J] = F1 + J ;
    }

    // FOR avec pas different
    FOR (I : 0 : 2 : 8) {
        Res = Res + I ;
    }

    /*
    ================================================================
     10. FOR IMBRIQUE (nested FOR)
    ================================================================
    */

    FOR (I : 0 : 1 : 4) {
        FOR (J : 0 : 1 : 4) {
            Res = I + J ;
        }
    }

    // Triple FOR imbrique
    FOR (I : 0 : 1 : 3) {
        FOR (J : 0 : 1 : 3) {
            FOR (K : 0 : 1 : 3) {
                Res = I + J + K ;
            }
        }
    }

    /*
    ================================================================
     11. WHILE IMBRIQUE (nested WHILE)
    ================================================================
    */

    I = 0 ;
    WHILE (I < 5) {
        J = 0 ;
        WHILE (J < 5) {
            Res = I * 5 + J ;
            J = J + 1 ;
        }
        I = I + 1 ;
    }

    /*
    ================================================================
     12. IF DANS WHILE
    ================================================================
    */

    I = 0 ;
    WHILE (I < N) {
        IF (T[I] > 0) {
            Res = Res + T[I] ;
        }
        ELSE {
            Res = Res - 1 ;
        }
        I = I + 1 ;
    }

    // IF avec condition composee dans WHILE
    I = 0 ;
    WHILE (I < N) {
        IF (T[I] > 0 && T[I] < 50) {
            T[I] = T[I] * 2 ;
        }
        I = I + 1 ;
    }

    /*
    ================================================================
     13. IF DANS FOR
    ================================================================
    */

    FOR (I : 0 : 1 : 9) {
        IF (T[I] > Max) {
            Max = T[I] ;
        }
    }

    FOR (I : 0 : 1 : 9) {
        IF (T[I] >= 0 && T[I] <= 9) {
            T[I] = T[I] + 1 ;
        }
        ELSE {
            T[I] = 0 ;
        }
    }

    /*
    ================================================================
     14. WHILE DANS FOR
    ================================================================
    */

    FOR (I : 0 : 1 : 4) {
        J = 0 ;
        WHILE (J < I) {
            Res = Res + J ;
            J = J + 1 ;
        }
    }

    /*
    ================================================================
     15. FOR DANS WHILE
    ================================================================
    */

    I = 1 ;
    WHILE (I <= 3) {
        FOR (J : 0 : 1 : 4) {
            MAT[J] = I * J ;
        }
        I = I + 1 ;
    }

    /*
    ================================================================
     16. IF IMBRIQUE DANS FOR IMBRIQUE
    ================================================================
    */

    FOR (I : 0 : 1 : 9) {
        FOR (J : 0 : 1 : 9) {
            IF (I == J) {
                Res = Res + 1 ;
            }
            ELSE {
                IF (I > J) {
                    Res = Res + 2 ;
                }
                ELSE {
                    Res = Res - 1 ;
                }
            }
        }
    }

    /*
    ================================================================
     17. IF IMBRIQUE DANS WHILE IMBRIQUE
    ================================================================
    */

    I = 0 ;
    WHILE (I < 5) {
        J = 0 ;
        WHILE (J < 5) {
            IF (I > J) {
                IF (Res > 0) {
                    Res = Res + I ;
                }
            }
            ELSE {
                Res = Res + J ;
            }
            J = J + 1 ;
        }
        I = I + 1 ;
    }

    /*
    ================================================================
     18. WRITE ET READ
    ================================================================
    */

    WRITE(A) ;
    WRITE(B) ;
    WRITE(Res) ;
    WRITE(F1) ;
    WRITE(T[0]) ;

    READ(A) ;
    READ(B) ;
    READ(N) ;

    // READ puis utilisation immediate
    READ(X) ;
    IF (X > 0) {
        Y = X * 2 ;
        WRITE(Y) ;
    }
    ELSE {
        WRITE(X) ;
    }

    /*
    ================================================================
     19. WRITE ET READ DANS BOUCLES
    ================================================================
    */

    

    FOR (I : 0 : 1 : 9) {
        WRITE(T[I]) ;
    }

    I = 0 ;
    WHILE (I < N) {
        IF (T[I] < 0) {
            T[I] = 0 ;
        }
        I = I + 1 ;
    }

    /*
    ================================================================
     20. ALGORITHME COMPLET : TRI A BULLES
    ================================================================
    */

    FOR (I : 0 : 1 : 8) {
        FOR (J : 0 : 1 : 8) {
            IF (T[J] > T[J]) {
                Res = T[J] ;
                T[J] = T[J] ;
                T[J] = Res ;
            }
        }
    }

    /*
    ================================================================
     21. ALGORITHME COMPLET : RECHERCHE DU MAX
    ================================================================
    */

    Max = T[0] ;
    FOR (I : 1 : 1 : 9) {
        IF (T[I] > Max) {
            Max = T[I] ;
        }
    }
    WRITE(Max) ;

    /*
    ================================================================
     22. ALGORITHME COMPLET : SOMME DES POSITIFS
    ================================================================
    */

    Res = 0 ;
    I = 0 ;
    WHILE (I < N) {
        IF (T[I] > 0) {
            Res = Res + T[I] ;
        }
        I = I + 1 ;
    }
    WRITE(Res) ;

    /*
    ================================================================
     23. EXPRESSIONS DANS CONDITIONS
    ================================================================
    */

    IF (A + B > C * D) {
        Res = A + B ;
    }

    IF ((A + B) * 2 <= (C + D) * 3) {
        Res = 1 ;
    }
    ELSE {
        Res = 0 ;
    }

    WHILE (I * 2 < N) {
        I = I + 1 ;
    }

    FOR (I : 0 : 1 : 9) {
        IF (T[I] * 2 > N + M) {
            T[I] = T[I] - 1 ;
        }
    }

END
